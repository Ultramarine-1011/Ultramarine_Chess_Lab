cmake_minimum_required(VERSION 3.24)
project(UltramarineChessLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
option(UCL_SANITIZERS "Enable address/undefined sanitizers (Clang/GCC)" OFF)
add_library(chess_library INTERFACE)
target_include_directories(chess_library SYSTEM INTERFACE third_party/chess-library)
add_library(ucl_core engine/src/evaluator.cpp engine/src/search.cpp engine/src/position.cpp)
target_include_directories(ucl_core PUBLIC engine/include)
target_link_libraries(ucl_core PUBLIC chess_library Threads::Threads)
if(MSVC)
  target_compile_options(ucl_core PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(ucl_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(UCL_SANITIZERS)
    target_compile_options(ucl_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ucl_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(ucl engine/src/main.cpp engine/src/uci.cpp)
target_link_libraries(ucl PRIVATE ucl_core)
target_compile_definitions(ucl PRIVATE UCL_VERSION="${PROJECT_VERSION}")
find_package(Git QUIET)
set(UCL_REVISION "unknown")
set(UCL_DIRTY "true")
if(GIT_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    OUTPUT_VARIABLE UCL_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  execute_process(COMMAND ${GIT_EXECUTABLE} status --porcelain WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    OUTPUT_VARIABLE UCL_GIT_STATUS OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(UCL_GIT_STATUS STREQUAL "")
    set(UCL_DIRTY "false")
  endif()
endif()
string(TOUPPER "${CMAKE_BUILD_TYPE}" UCL_BUILD_UPPER)
set(UCL_BUILD_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${UCL_BUILD_UPPER}}")
configure_file(engine/src/build_info.hpp.in generated/ucl/build_info.hpp @ONLY)
target_include_directories(ucl PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
install(TARGETS ucl RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  add_executable(ucl_tests tests/core_tests.cpp)
  target_link_libraries(ucl_tests PRIVATE ucl_core)
  add_test(NAME unit COMMAND ucl_tests unit)
  add_test(NAME search COMMAND ucl_tests search)
  add_test(NAME perft COMMAND ucl_tests perft)
  add_test(NAME uci COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_uci.py $<TARGET_FILE:ucl>)
  add_test(NAME experiments COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_experiments.py)
  set_tests_properties(experiments PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    ENVIRONMENT "PYTHONPATH=${CMAKE_SOURCE_DIR}/python")
  set_tests_properties(unit search perft uci experiments PROPERTIES TIMEOUT 60)
endif()
